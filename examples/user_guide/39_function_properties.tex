\begin{axdef}
modulo3 : \nat \surj \{0, 1, 2\}
\where
\forall n : \nat @ modulo3(n) = n \mod 3 \land \\
\t1 (\forall r : \{0, 1, 2\} @ \exists n : \nat @ modulo3(n) = r)
\end{axdef}
